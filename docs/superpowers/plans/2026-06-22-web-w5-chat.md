# Web W5 — Chat (Stream React SDK) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Player chat on web via `stream-chat-react` — channel list + conversation (DM/group/event), start-DM, and group/event "Open chat" entries — reusing `@padel/api` chat hooks, never gating the app on chat.

**Architecture:** A route-scoped `WebChatProvider` (mounts the Stream client + SDK CSS only on `/app/chat/**`) connecting the user via `useStreamToken`+`useMyProfile` (mirroring mobile's `StreamChatProvider`). Routes: `/app/chat` (two-pane list+conversation), `/app/chat/[cid]` (deep-link), `/app/chat/new` (start DM). Entries: a Messages nav item + group/event Open-chat buttons via `useEnsureChannel`.

**Tech Stack:** Next.js 16 App Router (client), React 19, `stream-chat` + `stream-chat-react`, `@padel/api`, `@padel/auth`, shadcn/ui, react-i18next.

- [ ] **Step 4: `WebChatProvider.tsx`**
```tsx
'use client';
import { type PropsWithChildren, useEffect } from 'react';
import { Chat } from 'stream-chat-react';
import { useMyProfile, useStreamToken } from '@padel/api';
import { useSession } from '@padel/auth';
import { streamClient } from '@/lib/streamClient';
import 'stream-chat-react/dist/css/v2/index.css';

export function WebChatProvider({ children }: PropsWithChildren) {
  const uid = useSession().session?.user.id;
  const tokenQ = useStreamToken();
  const profile = useMyProfile();

  useEffect(() => {
    const data = tokenQ.data;
    if (!uid || !data) {
      if (streamClient.userID) void streamClient.disconnectUser();
      return;
    }
    let cancelled = false;
    void (async () => {
      try {
        if (streamClient.userID === data.userId) return;
        if (streamClient.userID) await streamClient.disconnectUser();
        if (cancelled) return;
        await streamClient.connectUser(
          {
            id: data.userId,
            name: profile.data?.full_name ?? 'Player',
            image: profile.data?.avatar_url ?? undefined,
          },
          data.token,
        );
      } catch {
        /* connect failed; the chat surface shows its own error/retry */
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [uid, tokenQ.data, profile.data?.full_name, profile.data?.avatar_url]);

  if (!uid) return <>{children}</>;
  return <Chat client={streamClient}>{children}</Chat>;
}
```
(Note: `useMyProfile` row fields — confirm `full_name`/`avatar_url` exist; mobile uses exactly these.)

## Task 2: `/app/chat` two-pane (channel list + conversation)

**Files:** Create `apps/web/src/app/(app)/app/chat/page.tsx`; create `apps/web/src/components/chat/ChatShell.tsx`.

- [ ] **Step 1: `ChatShell.tsx`** — the reusable two-pane chat surface (used by both `/app/chat` and `/app/chat/[cid]`). Props `{ activeCid?: string }`. Uses `stream-chat-react`:
```tsx
'use client';
import { useEffect } from 'react';
import Link from 'next/link';
import { useT } from '@padel/i18n';
import {
  Channel,
  ChannelList,
  MessageInput,
  MessageList,
  Thread,
  Window,
  useChatContext,
} from 'stream-chat-react';
import { useStreamToken } from '@padel/api';
import { useSession } from '@padel/auth';
import { Button } from '@/components/ui/button';
import { Skeleton } from '@/components/ui/skeleton';

function ActiveChannelSetter({ cid }: { cid: string }) {
  const { client, setActiveChannel } = useChatContext();
  useEffect(() => {
    let cancelled = false;
    void (async () => {
      const [type, id] = cid.split(':');
      if (!type || !id) return;
      const ch = client.channel(type, id);
      await ch.watch();
      if (!cancelled) setActiveChannel(ch);
    })();
    return () => { cancelled = true; };
  }, [cid, client, setActiveChannel]);
  return null;
}

export function ChatShell({ activeCid }: { activeCid?: string }) {
  const { t } = useT('chat');
  const uid = useSession().session?.user.id;
  const tokenQ = useStreamToken();

  if (tokenQ.isError) {
    return (
      <div className="flex flex-col items-start gap-3 p-6">
        <p className="text-sm text-muted-foreground">{t('chatUnavailable')}</p>
        <Button variant="outline" onClick={() => tokenQ.refetch()}>{t('retry')}</Button>
      </div>
    );
  }
  if (!uid || !tokenQ.data) return <Skeleton className="m-6 h-[70vh]" />;

  return (
    <div className="flex h-[calc(100vh-3.5rem)] flex-col">
      <div className="flex items-center justify-between border-b p-3">
        <h1 className="text-lg font-semibold">{t('title')}</h1>
        <Button asChild size="sm"><Link href="/app/chat/new">{t('newMessage')}</Link></Button>
      </div>
      <div className="grid min-h-0 flex-1 grid-cols-[320px_1fr]">
        <div className="min-h-0 overflow-y-auto border-r">
          <ChannelList
            filters={{ type: 'messaging', members: { $in: [uid] } }}
            sort={{ last_message_at: -1 }}
            options={{ state: true, watch: true, presence: true }}
          />
        </div>
        <div className="min-h-0">
          {activeCid ? <ActiveChannelSetter cid={activeCid} /> : null}
          <Channel>
            <Window>
              <MessageList />
              <MessageInput />
            </Window>
            <Thread />
          </Channel>
        </div>
      </div>
    </div>
  );
}
```
VERIFY against the installed `stream-chat-react` 14.x API: `ChannelList` props (`filters`/`sort`/`options`), `Channel` rendering the active channel from context when no `channel` prop is passed, `useChatContext` exposing `client` + `setActiveChannel`. If 14.x differs (e.g. `Channel` requires an explicit `channel` prop), adapt: track the active channel via `useChatContext().channel` and pass it, or use the `ChannelList`'s default selection behavior. Keep the empty/"select a conversation" affordance — Stream's default `EmptyStateIndicator` covers "no channels"; if you want the custom `noChannels`/`selectConversation` copy, pass `EmptyStateIndicator` props (optional).

- [ ] **Step 2: `new/page.tsx`** — start a DM from the people the user follows:
```tsx
'use client';
import { useState } from 'react';
import { useRouter } from 'next/navigation';
import { useT } from '@padel/i18n';
import { useFollowing } from '@padel/api';
import { useSession } from '@padel/auth';
import { streamClient } from '@/lib/streamClient';
import { Avatar, AvatarFallback, AvatarImage } from '@/components/ui/avatar';
import { Button } from '@/components/ui/button';
import { Card } from '@/components/ui/card';
import { Skeleton } from '@/components/ui/skeleton';
import { avatarUrl } from '@/lib/upload';

export default function NewChatPage() {
  const { t } = useT('chat');
  const router = useRouter();
  const uid = useSession().session?.user.id;
  const following = useFollowing(uid);
  const [busy, setBusy] = useState(false);

  const onPick = async (otherId: string) => {
    if (!uid || busy) return;
    setBusy(true);
    try {
      const ch = streamClient.channel('messaging', { members: [uid, otherId] });
      await ch.watch();
      router.push(`/app/chat/${encodeURIComponent(ch.cid)}`);
    } catch {
      setBusy(false);
    }
  };

  return (
    <div className="flex flex-col gap-4 p-6">
      <h1 className="text-lg font-semibold">{t('newMessage')}</h1>
      {following.isLoading ? (
        <Skeleton className="h-40 w-full" />
      ) : (following.data ?? []).length === 0 ? (
        <p className="text-sm text-muted-foreground">{t('noFollowing')}</p>
      ) : (
        <Card className="divide-y p-0">
          {(following.data ?? []).map((u) => {
            const name = u.profiles?.full_name ?? '—';
            return (
              <button
                key={u.profiles?.id ?? u.followee_id}
                type="button"
                disabled={busy}
                onClick={() => onPick(u.profiles?.id ?? u.followee_id)}
                className="flex w-full items-center gap-3 px-4 py-3 text-left hover:bg-muted/50"
              >
                <Avatar className="size-9">
                  <AvatarImage src={avatarUrl(u.profiles?.avatar_url) ?? undefined} />
                  <AvatarFallback>{name.slice(0, 2).toUpperCase()}</AvatarFallback>
                </Avatar>
                <span className="truncate text-sm font-medium">{name}</span>
              </button>
            );
          })}
        </Card>
      )}
    </div>
  );
}
```
VERIFY the `useFollowing` row shape (`packages/api/src/profile/queries.ts`): the field holding the other user's id + profile (e.g. `followee_id` + `profiles:{id, full_name, avatar_url}`). Adapt the `key`/`onPick`/name access to the real shape and report what it is. (Reuse exactly how the W1 following list renders rows.)
